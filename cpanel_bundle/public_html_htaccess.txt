# ==============================================================================
# LiteSpeed / Apache Rewrite Rules for b-Smart Online
# File location on cPanel: /home/bsmarton/public_html/.htaccess
# ==============================================================================
# This ensures that when anyone opens a card link at either:
#   https://b-smart-online.co.za/card/your-name
# or
#   https://b-smart-online.co.za/app/card/your-name
# LiteSpeed web server routes it seamlessly without showing 404!
# ==============================================================================

RewriteEngine On
RewriteBase /

# 1. Forward all direct card links to the Node.js /app application
RewriteRule ^card/(.*)$ /app/card/$1 [R=301,L]
